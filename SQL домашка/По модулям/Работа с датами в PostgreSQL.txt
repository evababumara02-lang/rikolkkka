select event_id, event_type, 
        DATE '2001-01-01'+ (start_day-1) as start_day,
        DATE '2001-01-01' + (end_day-1) as end_day
 from events e 
 where end_day is not null;

 -- задача 2
 select event_id, event_type, 
 DATE '2001-01-01'+ (start_day-1) as start_day
 from events e 
 where start_day>=1 and start_day <= 31;
 
 -- задача 3
 
 select event_id, event_type, 
        DATE '2001-01-01'+ (start_day-1) as start_day,
        extract (day from date '2001-01-01'+ (start_day-1)) as start_date 
 from events 
 order by start_date, event_id
 
 -- задача 4
 select event_id, event_type, end_day-start_day + 1 as duration
 from events e 
 where end_day is not null
 order by duration desc , event_id limit 3
 
 -- задача 5
 select event_id, event_type, risk_level,
         DATE '2001-01-01'+ (start_day-1) as start_day
 from events e 
 where  DATE '2001-01-01'+ (start_day-1) < date '2001-01-05' and risk_level >= 7;
 
 -- задача 6
 select event_id, event_type,
         DATE '2001-01-01'+ (start_day-1) as start_day,
         extract (dow from date '2001-01-01'+ (start_day-1)) as dow
from events
where extract (dow from date '2001-01-01'+ (start_day-1)) in (0,6)
order by start_day, event_id;
