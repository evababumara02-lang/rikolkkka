-- задача 1
with cte as (select *, end_day-start_day as duration from events e 
             where end_day is not null),
     cte2 as ( select max ( duration) as max_dur from cte)
select event_id, duration 
from cte 
join cte2 
on cte.duration = cte2.max_dur;

-- задача 2
with cte AS (select * from events e 
              join locations l on e.location_id = l.location_id ),
      cte2 as ( select cte.name, count (*) as cnt from cte
               group by cte.name),
       cte3 as ( select max (cnt) as max_cnt from cte2)
select cte2.name, cnt
from cte2
join cte3
on cte2.cnt=cte3.max_cnt;

-- задача 3
with t1 as ( select distinct ep.character_id, count (*) as cnt from event_participants ep  
            join events e on ep.event_id=e.event_id 
            where e. risk_level >= 8 group by character_id ) 
select character_id , cnt 
from t1
where cnt>1;

-- задача 4
with overall_risk as (select avg (risk_level) overall_risk from events e),
     quests_risk as (select q.quest_id,q.quest_name, avg ( risk_level) avg_quest_risk from quests q 
            join quest_events qe on q.quest_id = qe.quest_id
            join events e on qe.event_id = e.event_id 
            group by q.quest_id, q.quest_name)
select quest_id ,quest_name , avg_quest_risk 
from quests_risk 
cross join overall_risk 
where avg_quest_risk > overall_risk;

-- задача 5
with cte as ( select c."name"  , count (c.character_id ) as cnt_character 
              from "characters" c 
             join event_participants ep on c.character_id = ep.character_id
              group by c."name" )
select name , cnt_character 
from cte  
order by cnt_character desc 
limit 5;

-- задача 10
 with avg_location as (select l.location_id, l."name"   , avg ( end_day-start_day) as avg_duration from locations l 
              join events e  on e.location_id = l.location_id, where end_day  is not null 
              group by l.location_id, l."name" ),
      overall_avg as ( select  avg (end_day-start_day) from events where end_day is not null) 
  select name, avg_duration  from avg_location 
  cross join overall_avg 
  where avg_location > overall_avg
