-- задача 1
SELECT e.event_type, l.name location_name, l.TYPE location_type FROM events e  
left join locations l on e.location_id = l.location_id

-- задача 2
select distinct  e.event_id, e.event_type from events e  
join event_participants ep on ep.event_id = e.event_id 
join "characters" c on c.character_id = ep.character_id 
where c.name='Donkey'

-- задача 3
select q.quest_id, q.quest_name, q.success, count (qe.event_id) events_in_quests from quests q 
left join quest_events qe on q.quest_id=qe.quest_id 
group by q.quest_id, q.quest_name, q.success

-- задача 4
select distinct e.event_id, ep.role_in_event from events e 
inner join event_participants ep on e.event_id=ep.event_id 
where ep.role_in_event = 'victim'

-- задача 5
select ep.event_id, l.name location_name, c.name characters_name,  ep.role_in_event from events e 
inner join locations l  on e.location_id =l.location_id
inner join event_participants ep on ep.event_id=e.event_id 
inner join "characters" c on c.character_id =ep.character_id 

-- задача 6
select distinct l.name from locations l 
inner  join events e on e.location_id=l.location_id 
inner join event_participants ep on ep.event_id=e.event_id 
inner join "characters" c on c.character_id=ep.character_id  
where c.name='Shrek'
 
-- задача 7

select e.event_id, e.event_type, c.name character_name, ep.role_in_event from "characters" c 
join event_participants ep on ep.character_id =c.character_id 
join events e  on e.event_id=ep.event_id 
where c. alignment = 'evil' order by e.event_id, c.name

-- задача 8
select q.quest_id, e.event_id from quests q 
inner join quest_events qe on qe.quest_id =q.quest_id 
inner join events e on e.event_id=qe.event_id 
order by e.risk_level >=8

-- задача 9
select e.event_id,COUNT (ep.character_id) unique_character_id from events e 
inner join event_participants ep on ep.event_id =e.event_id group by e.event_id

--задание 10
select ep.character_id, c.name from "characters" c 
left join event_participants ep on ep.character_id = c.character_id 
where ep.event_id is null
